# ============================================================================
# 🔴 OBRIGATÓRIAS - Sistema não funciona sem estas
# ============================================================================

# --- Embeddings (Voyage AI) ---
# Obtenha em: https://dash.voyageai.com/
VOYAGE_API_KEY=your_voyage_api_key_here

# --- Generator/Chat (escolha UM) ---
# Claude (Anthropic) - Recomendado
# Obtenha em: https://console.anthropic.com/
ANTHROPIC_API_KEY=your_anthropic_api_key_here

# OU GPT-4 (OpenAI)
# Obtenha em: https://platform.openai.com/api-keys
# OPENAI_API_KEY=your_openai_api_key_here

# OU Gemini (Google)
# Obtenha em: https://makersuite.google.com/app/apikey
# GOOGLE_API_KEY=your_google_api_key_here


# ============================================================================
# 🟡 RECOMENDADAS - Melhoram significativamente a qualidade
# ============================================================================

# --- Reranker (Cohere) ---
# Melhora a relevância dos resultados em 30-40%
# Obtenha em: https://dashboard.cohere.com/api-keys
COHERE_API_KEY=your_cohere_api_key_here

# --- Alternativas de Reranker (se não usar Cohere) ---
# Jina AI Reranker
# JINA_API_KEY=your_jina_api_key_here

# Voyage Reranker (usa a mesma VOYAGE_API_KEY acima)
# Não precisa configurar nada adicional


# ============================================================================
# 🟢 OPCIONAIS - Features Avançadas
# ============================================================================

# --- Unified Consulting Ingestor: Visual Analysis Premium ---
# Para extrair gráficos/tabelas de PPTX/PDF com alta precisão
# Obtenha em: https://contextual.ai/
# CONTEXTUAL_API_KEY=your_contextual_api_key_here
# CONTEXTUAL_API_URL=https://api.contextual.ai/v1/documents

# --- Unified Consulting Ingestor: Docling API (Self-Hosted) ---
# Se você tem um servidor Docling próprio para OCR
# DOCLING_API_URL=http://your-docling-server:5000

# --- Google Drive Reader ---
# Para importar arquivos diretamente do Google Drive
# GOOGLE_DRIVE_CREDENTIALS=/path/to/service-account-key.json

# --- Apache Tika (OCR Avançado) ---
# Para processar PDFs escaneados e imagens
# TIKA_SERVER_URL=http://localhost:9998

# --- Named Vectors (Experimental) ---
# Habilita vetores especializados (concept, company, sector)
# ENABLE_NAMED_VECTORS=true

# --- spaCy NER/ETL ---
# Modelo para extração de entidades (PT ou EN)
# SPACY_MODEL=pt_core_news_sm


# ============================================================================
# ⚙️ CONFIGURAÇÃO WEAVIATE (Railway auto-configura)
# ============================================================================

# WEAVIATE_URL_VERBA=http://weaviate:8080
# WEAVIATE_API_KEY_VERBA=your_weaviate_api_key
# WEAVIATE_HTTP_HOST=weaviate-instance.weaviate.network
# WEAVIATE_HTTP_PORT=443
